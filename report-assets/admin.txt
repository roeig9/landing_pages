<script>
(function () {
  const A = window.__ADMIN__, d = window.__report;
  const $ = id => document.getElementById(id);
  const labels = { draft: 'טיוטה. הלקוח עוד לא ראה את הדוח.', ready: 'הדוח נשלח ללקוח. שינויים שתשמור יופיעו אצלו מיד.' };
  $('admSt').textContent = 'מצב עריכה · ' + (labels[A.status] || A.status) + ' לחיצה על כל טקסט עם מסגרת מאפשרת לתקן אותו.';
  if (A.problems) { $('admPr').hidden = false; $('admPr').textContent = 'הבדיקה האוטומטית מצאה: ' + A.problems; }
  if (A.status === 'ready') $('admOk').textContent = 'שמירה ושליחה חוזרת ללקוח';

  const fields = document.querySelectorAll('[data-f]');
  fields.forEach(el => {
    try { el.contentEditable = 'plaintext-only'; } catch (e) { el.contentEditable = 'true'; }
    if (el.contentEditable !== 'plaintext-only') el.contentEditable = 'true';
    el.dataset.orig = el.innerText.trim();
    el.addEventListener('input', () => el.classList.toggle('changed', el.innerText.trim() !== el.dataset.orig));
  });

  function edits() {
    const e = {};
    fields.forEach(el => { e[el.dataset.f] = el.innerText.replace(/ /g, ' ').trim(); });
    return e;
  }
  async function send(action) {
    const note = $('admNote').value.trim();
    if (action === 'rewrite' && !note) { $('admMsg').textContent = 'כדאי לכתוב בהערות מה לשנות, ואז ללחוץ על כתיבה מחדש.'; return; }
    if (action === 'approve' && !confirm(A.status === 'ready' ? 'לשמור ולשלוח שוב את הקישור ללקוח?' : 'לאשר ולשלוח את הדוח ללקוח?')) return;
    document.querySelectorAll('#adm button').forEach(b => b.disabled = true);
    $('admMsg').textContent = action === 'rewrite' ? 'כותב מחדש, זה לוקח כחצי דקה...' : 'שומר...';
    try {
      const res = await fetch(location.origin + '/webhook/report-admin', {
        method: 'POST', headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ id: A.id, k: A.k, action, edits: edits(), comment: note }),
      });
      const r = await res.json().catch(() => ({}));
      if (!res.ok || !r.ok) throw new Error(r.message || ('HTTP ' + res.status));
      $('admMsg').textContent = r.message || 'נשמר';
      if (action !== 'approve') setTimeout(() => location.reload(), 900);
      else document.querySelectorAll('#adm button').forEach(b => b.disabled = false);
    } catch (err) {
      $('admMsg').textContent = 'לא הצליח: ' + err.message;
      document.querySelectorAll('#adm button').forEach(b => b.disabled = false);
    }
  }
  $('admSave').onclick = () => send('save');
  $('admRew').onclick = () => send('rewrite');
  $('admOk').onclick = () => send('approve');
})();
</script>
