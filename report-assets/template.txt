<script>
(function () {
  const d = JSON.parse(document.getElementById("report-data").textContent);
  window.__report = d;
  const leading = d.scenario === "leading";
  const F = p => ` data-f="${p}"`;
  const esc = s => String(s).replace(/[&<>"]/g, c => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;" }[c]));

  // In the "leading" scenario the map is coloured by top 3, otherwise by top 10.
  function cls(r, i) {
    let c = r == null ? "out" : r <= 3 ? "t3" : r <= 10 ? (leading ? "t20" : "t10") : r <= 20 ? "t20" : "out";
    return i === 24 ? c + " home" : c;
  }
  const map = (m, mi) => `
    <div class="map">
      <h3>${esc(m.service)}</h3>
      <div class="grid" role="img" aria-label="${esc(m.line)}">
        ${m.ranks.map((r, i) => `<span class="${cls(r, i)}">${r == null ? "20+" : r}</span>`).join("")}
      </div>
      <p${F('maps.' + mi + '.line')}>${esc(m.line)}</p>
    </div>`;

  const scopeN = d.scope_points || 49;
  const km = x => x == null ? "–" : (x < 1 ? Math.round(x * 1000) + " מ'" : x + ' ק"מ');
  const comp = d.competitors.map(c => `<tr><td>${esc(c.name)}</td><td class="num">${km(c.distance_km)}</td><td class="num">${c.reviews ?? "–"} · ${c.rating != null ? c.rating.toFixed(1) : "–"}</td><td class="num">${c.points_ahead} מתוך ${scopeN}</td></tr>`).join("");

  document.getElementById("report").innerHTML = `
    <header>
      <h1>${esc(d.owner_first_name)}, ככה רואים את ${esc(d.business_name)} בגוגל</h1>
      <p class="meta">בדיקה מ-${esc(d.date)} · 49 נקודות סביב העסק, ${d.spacing_km} ק"מ בין נקודה לנקודה</p>
    </header>

    <p class="summary"${F('summary')}>${esc(d.summary)}</p>

    <section aria-labelledby="h-maps">
      <h2 id="h-maps">איפה מוצאים אותך</h2>
      <p class="lead">העסק במרכז המפה, במסגרת. כל משבצת היא נקודה במרחק ${d.spacing_km} ק"מ נוספים ממנו, והמספר הוא המקום שלך בגוגל מפות למי שמחפש משם את השירות שבכותרת המפה. החיפוש נעשה בדיוק במילים האלה, בלי שם עיר.</p>
      <div class="maps">${d.maps.map(map).join("")}</div>
      <div class="legend">
        <span><i class="t3"></i>מקום 1-3</span>
        <span><i class="${leading ? "t20" : "t10"}"></i>מקום 4-10</span>
        ${leading ? "" : '<span><i class="t20"></i>מקום 11-20</span>'}
        <span><i class="out"></i>לא בעשרים הראשונים</span>
      </div>
    </section>

    ${d.competitors && d.competitors.length ? `<section aria-labelledby="h-comp">
      <h2 id="h-comp">${esc(d.competitors_title)}</h2>
      <table>
        <thead><tr><th>עסק</th><th>מרחק ממך</th><th>ביקורות · דירוג</th><th>מופיע לפני העסק ב-</th></tr></thead>
        <tbody>${comp}<tr class="you"><td>${esc(d.business_name)}</td><td></td><td class="num">${d.you.reviews ?? "–"} · ${d.you.rating != null ? d.you.rating.toFixed(1) : "–"}</td><td></td></tr></tbody>
      </table>
      ${d.competitors_scope ? `<p class="lead">${esc(d.competitors_scope)}</p>` : ""}
      <p class="note"${F('competitors_note')}>${esc(d.competitors_note || "")}</p>
    </section>` : ""}

    <section aria-labelledby="h-why">
      <h2 id="h-why">למה זה קורה</h2>
      <ul class="findings">${d.findings.map((f, i) => `<li><strong${F('findings.' + i + '.title')}>${esc(f.title)}</strong><span${F('findings.' + i + '.detail')}>${esc(f.detail)}</span></li>`).join("")}</ul>
    </section>

    <section aria-labelledby="h-do">
      <h2 id="h-do">מה צריך לעשות</h2>
      <ol class="actions">${d.actions.map((a, i) => `<li><div><strong${F('actions.' + i + '.title')}>${esc(a.title)}</strong><span${F('actions.' + i + '.detail')}>${esc(a.detail)}</span><br><span class="tag ${a.type}">${a.type === "self" ? "אפשר לבד, היום" : "מערכת שמקימים פעם אחת"}</span></div></li>`).join("")}</ol>
      ${d.after_verification && d.after_verification.length ? `
      <div class="after">
        <h3>מיד אחרי האימות</h3>
        <ul>${d.after_verification.map((x, i) => `<li${F('after_verification.' + i)}>${esc(x)}</li>`).join("")}</ul>
      </div>` : ""}
    </section>

    ${d.description ? `
    <section aria-labelledby="h-desc">
      <h2 id="h-desc">תיאור עסק מוכן להדבקה</h2>
      <p class="lead">אפשר לשנות כל מילה. בלי טלפון או קישורים, הם כבר מופיעים בפרופיל.</p>
      <div class="copy" id="desc"${F('description')}>${esc(d.description)}</div>
      <button class="copy-btn" type="button" id="copyBtn">העתקת התיאור</button>
    </section>` : ""}

    <section class="cta" aria-labelledby="h-next">
      <h2 id="h-next">רוצה שנעבור על זה יחד?</h2>
      <p>שיחה של 20 דקות: נעבור על המפות ונחליט מה מקימים קודם. בעוד חודש נריץ את אותה בדיקה ונראה מה השתנה.</p>
      <a class="btn" href="${esc(d.cta_url)}" target="_blank" rel="noopener">לתיאום שיחה</a>
      <button class="pdf" type="button" onclick="window.print()">הורדה כ-PDF</button>
    </section>`;

  const btn = document.getElementById("copyBtn");
  if (btn) btn.addEventListener("click", async () => {
    try { await navigator.clipboard.writeText(d.description); btn.textContent = "הועתק"; }
    catch (e) { btn.textContent = "סמנו והעתיקו ידנית"; }
  });
})();
</script>
